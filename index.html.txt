<script>
  const totalItems = 7;
  let collectedCount = 0;
  const collectedSet = new Set();

  function collectItem(id, icon, element) {
    if (collectedSet.has(id)) return;

    collectedSet.add(id);
    collectedCount++;

    // 隱藏房間內的物品
    element.style.display = 'none';

    // 亮起物品欄
    const slot = document.getElementById(`slot-${id}`);
    if (slot) {
      slot.classList.add('collected');
    }

    // 更新 UI 文字
    const statusText = document.getElementById('status-text');
    statusText.innerText = `找到派對必備物品: ${collectedCount} / ${totalItems}`;

    if (collectedCount === totalItems) {
      statusText.innerText = "物品已集齊！點擊房門前往派對吧！";
      statusText.style.color = "#ff477e";
      document.getElementById('door').style.background = '#e0a96d';
    }
  }

  function tryOpenDoor() {
    if (collectedCount < totalItems) {
      alert(`還沒準備好喔！還缺少 ${totalItems - collectedCount} 件物品才能出門。`);
    } else {
      document.getElementById('ending-modal').style.display = 'flex';
    }
  }
</script>
